\begin{abstract}
This monograph records, in Lean~4 against Mathlib, a precise formal statement of Fefferman's Statement~(B) on global smoothness of the unforced incompressible Navier--Stokes equations on $\mathbb{T}^3=\mathbb{R}^3/\mathbb{Z}^3$, together with an exact ledger of the analytic inputs that the Frohmanian symplectic tether does \emph{not} supply. Statement~(B) is \emph{not} proved here. What is proved is a reduction: Statement~(B) follows from the classical local theory with the Beale--Kato--Majda continuation criterion and from the open hypothesis that the Beale--Kato--Majda integral is finite on every finite interval of existence. We also record the two-dimensional clock test: constant vortex direction ($\sin\varphi\equiv 0$) coexists with linear growth $\|\phi(t)\|_{L^\infty}=M_0^2\,t$ on steady Euler equilibria.
\end{abstract}

\section{Statement (B)}
Following the official problem description of Fefferman~\cite{Fef}, fix $\nu>0$ and let $u^\circ$ be a smooth, divergence-free vector field on $\mathbb{R}^3$ with $u^\circ(x+e_j)=u^\circ(x)$ for $j=1,2,3$. Take $f\equiv 0$.

\begin{quote}
\textbf{Statement (B).} There exist smooth functions $p(x,t)$, $u_i(x,t)$ on $\mathbb{R}^3\times[0,\infty)$ such that
\[
\partial_t u_i+\sum_{j=1}^3 u_j\,\partial_j u_i = \nu\,\Delta u_i-\partial_i p,\qquad \operatorname{div}u=0,\qquad u(x,0)=u^\circ(x),
\]
and $u$, $p$ are $1$-periodic in each spatial coordinate.
\end{quote}

In Lean this is the proposition \texttt{Clay.StatementB.FeffermanStatementB} (file \texttt{lean/Clay/StatementB.lean}). Space is $\mathbb{R}^3$ realised as Euclidean space on $\mathrm{Fin}\,3$; partial derivatives are Fréchet derivatives along the standard basis; the time derivative is taken within $[0,\infty)$; smoothness is $C^\infty$ on $[0,\infty)\times\mathbb{R}^3$. No component of the statement is opaque or trivial.

\section{The proved reduction}
Write $\omega=\operatorname{curl}u$. Consider the propositions
\begin{itemize}
  \item \texttt{PeriodicMaximalSolutionWithBKM} (classical; Kato, Fujita--Kato, Beale--Kato--Majda~\cite{BKM}; not yet formalized): for all admissible data there are $T^\ast\in(0,\infty]$ and a smooth solution on $[0,T^\ast)$ such that $T^\ast<\infty$ implies $\int_0^{T^\ast}\|\omega(t)\|_{L^\infty}\,dt=\infty$;
  \item \texttt{PeriodicBKMIntegrability} (\textbf{open}; the Clay core): every smooth solution on a finite interval $[0,T)$ satisfies $\int_0^{T}\|\omega(t)\|_{L^\infty}\,dt<\infty$.
\end{itemize}

\newtheorem{theorem}{Theorem}
\begin{theorem}[\texttt{feffermanStatementB\_of\_bkm}, kernel-checked]
\texttt{PeriodicMaximalSolutionWithBKM} and \texttt{PeriodicBKMIntegrability} together imply Statement~(B).
\end{theorem}

\begin{theorem}[\texttt{bkmIntegralFinite\_of\_holographic}, kernel-checked]
If $\|\omega(t)\|_{L^\infty}$ is bounded on every finite interval of existence, then \texttt{PeriodicBKMIntegrability} holds.
\end{theorem}

The second hypothesis of the first theorem is open and is equivalent in strength to Statement~(B) itself. Consequently the reduction does not prove Statement~(B).

\section{Ledger of missing inputs}
The file \texttt{lean/Clay/Gaps/MissingInputs.lean} states each former placeholder as a precise proposition:
\begin{itemize}
  \item \texttt{metriplecticControlsEnstrophyL2}: $\sup_{[0,T)}\int_{\mathbb{T}^3}|\omega|^2<\infty$ for every solution on a finite interval (open);
  \item \texttt{czGivesBMONotLinf}: no constant $C$ gives $\|\nabla v\|_{L^\infty}\le C\|\operatorname{curl}v\|_{L^\infty}$ for smooth periodic divergence-free $v$ (classical, not formalized);
  \item \texttt{spikeObstruction}: no constant $C$ gives $\|f\|_{L^\infty}\le C\|f\|_{L^6(\mathbb{T}^3)}$ for smooth periodic $f$ (classical, not formalized);
  \item \texttt{holographicNotAZFCBound}: the uniform ceiling $\sup_{[0,T)}\|\omega\|_{L^\infty}<\infty$ (open);
  \item \texttt{depletedStretching}: $\omega\cdot S\omega\le C|\omega|^2$ uniformly on finite intervals (open).
\end{itemize}
The former trivial definitions are retained under the suffix \texttt{\_placeholder} and marked deprecated. Kernel-checked elementary results include the identity $\Pi_u u=0$ for the pointwise tether projection, the unboundedness of the history integral $m^2t$ for $m>0$, and the conditional Riccati rearrangement.
